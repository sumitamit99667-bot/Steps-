<script>
(function () {
  "use strict";

  const links = {
    youtube1: "https://youtube.com/@bot_fathers?si=ryU8gMju1AOyjK4c",
    youtube2: "https://youtube.com/@bot_fathers786?si=VIZzKdwhJBFAbS_H",
    telegram: "https://t.me/AMITOURNAMENTFATHER",
    final: "https://t.me/AMITOURNAMENTFATHER"
  };

  const STORAGE_KEY = "botFatherStepsCompleted_session_v1";

  const stepIds = ["step1", "step2", "step3"];
  const statusIds = ["status1", "status2", "status3"];

  const stepLinks = [
    links.youtube1,
    links.youtube2,
    links.telegram
  ];

  const completed = new Set();
  const loadingSteps = new Set();

  const progressText = document.getElementById("progressText");
  const progressFill = document.getElementById("progressFill");
  const getLinkButton = document.getElementById("getLink");
  const message = document.getElementById("message");

  // Restore progress in the current tab session.
  try {
    const saved = sessionStorage.getItem(STORAGE_KEY);

    if (saved) {
      const parsed = JSON.parse(saved);

      if (Array.isArray(parsed)) {
        parsed.forEach(function (value) {
          if ([0, 1, 2].includes(value)) {
            completed.add(value);
          }
        });
      }
    }
  } catch (error) {}

  function saveProgress() {
    try {
      sessionStorage.setItem(
        STORAGE_KEY,
        JSON.stringify(Array.from(completed))
      );
    } catch (error) {}
  }

  function updateUI() {
    const count = completed.size;

    progressText.textContent = count + " / 3 COMPLETED";
    progressFill.style.width = (count / 3 * 100) + "%";

    stepIds.forEach(function (id, index) {
      const card = document.getElementById(id);
      const status = document.getElementById(statusIds[index]);

      card.classList.toggle("completed", completed.has(index));

      if (!loadingSteps.has(index)) {
        status.classList.remove("loading-text");
        status.textContent = completed.has(index) ? "✓ DONE" : "OPEN ↗";
      }
    });

    if (count === 3) {
      getLinkButton.disabled = false;
      getLinkButton.classList.add("unlocked");
      getLinkButton.textContent = "🔓 GET YOUR LINK";
      message.textContent = "All steps completed! Your link is unlocked.";
      message.style.color = "#00ff99";
    } else {
      getLinkButton.disabled = true;
      getLinkButton.classList.remove("unlocked");
      getLinkButton.textContent = "🔒 COMPLETE ALL STEPS";

      message.textContent = loadingSteps.size
        ? "Loading is running. Your step will complete in 10 seconds."
        : "Complete " + (3 - count) + " more step(s) to unlock your link.";

      message.style.color = "#9ca8bd";
    }
  }

  function openStep(index) {
    const card = document.getElementById(stepIds[index]);
    const status = document.getElementById(statusIds[index]);

    if (loadingSteps.has(index)) return;

    if (completed.has(index)) {
      window.open(stepLinks[index], "_blank");
      return;
    }

    /*
      Open the link immediately on the click.
      The 10-second timer continues on the original page.
    */
    const newTab = window.open(stepLinks[index], "_blank");

    if (!newTab) {
      // Pop-up blocked: use the current tab instead.
      window.location.href = stepLinks[index];
      return;
    }

    newTab.opener = null;

    loadingSteps.add(index);
    card.classList.add("loading");
    status.classList.add("loading-text");

    let seconds = 0;
    status.innerHTML = '<span class="loading-spinner"></span>0%';

    message.textContent = "Loading... Please wait 10 seconds.";
    message.style.color = "#c4a2ff";

    const timer = setInterval(function () {
      seconds++;

      status.innerHTML =
        '<span class="loading-spinner"></span>' + (seconds * 10) + "%";

      if (seconds >= 10) {
        clearInterval(timer);

        loadingSteps.delete(index);
        completed.add(index);

        saveProgress();

        card.classList.remove("loading");
        updateUI();
      }
    }, 1000);
  }

  stepIds.forEach(function (id, index) {
    document.getElementById(id).addEventListener("click", function () {
      openStep(index);
    });
  });

  getLinkButton.addEventListener("click", function () {
    if (completed.size !== 3) return;
    window.location.href = links.final;
  });

  updateUI();
  window.addEventListener("pageshow", updateUI);
})();
</script>
